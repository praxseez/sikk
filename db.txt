/* ============================================================
   НАСТРОЙКА MS SQL ДЛЯ DJANGO-ПРОЕКТА SIKK
   Выполнять под администратором (sa) в SSMS или Azure Data Studio
   ============================================================ */

/* ------------------------------------------------------------
   1. СОЗДАНИЕ ЛОГИНА НА УРОВНЕ СЕРВЕРА
   ------------------------------------------------------------ */
CREATE LOGIN sikk_user 
WITH PASSWORD = 'sikk_password',
     CHECK_POLICY = OFF;
GO

/* ------------------------------------------------------------
   2. СОЗДАНИЕ БАЗЫ ДАННЫХ
   ------------------------------------------------------------ */
CREATE DATABASE sikk_db;
GO

/* ------------------------------------------------------------
   3. НАЗНАЧЕНИЕ ЛОГИНА ВЛАДЕЛЬЦЕМ БАЗЫ
   ВАЖНО: делать это ДО создания пользователя внутри базы,
   иначе получите ошибку 15110.
   ------------------------------------------------------------ */
ALTER AUTHORIZATION ON DATABASE::sikk_db TO sikk_user;
GO

/* ------------------------------------------------------------
   4. ПРОВЕРКА, ЧТО ВЛАДЕЛЕЦ СМЕНИЛСЯ
   В owner_name должно быть sikk_user, а не ваша Windows-учётка
   ------------------------------------------------------------ */
SELECT name, SUSER_SNAME(owner_sid) AS owner_name
FROM sys.databases
WHERE name = 'sikk_db';
GO

/* ------------------------------------------------------------
   5. (ОПЦИОНАЛЬНО) ЯВНЫЕ ПРАВА НА ЧТЕНИЕ/ЗАПИСЬ
   Для Django обычно не требуется, так как владелец базы
   автоматически получает все права через псевдоним dbo.
   Но можно добавить для явности.
   ------------------------------------------------------------ */
USE sikk_db;
GO
